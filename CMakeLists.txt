cmake_minimum_required(VERSION 3.24)
project(WebCaptureMobile VERSION 0.3.0 LANGUAGES C CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows 11 x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(JUCE_DIR "" CACHE PATH "JUCE source")
set(CEF_ROOT "" CACHE PATH "CEF minimal distribution")
add_subdirectory("${JUCE_DIR}" juce)
juce_add_plugin(WebCaptureMobile COMPANY_NAME "PAPIELGUAPO" PLUGIN_MANUFACTURER_CODE Wcap PLUGIN_CODE Wm02 FORMATS VST3 Standalone PRODUCT_NAME "Joseador" IS_SYNTH TRUE EDITOR_WANTS_KEYBOARD_FOCUS TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(WebCaptureMobile PRIVATE Source/Plugin.cpp Source/Bridge.h Source/Session.h Source/Browser.h)
target_compile_options(WebCaptureMobile PRIVATE /utf-8)
target_compile_definitions(WebCaptureMobile PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(WebCaptureMobile PRIVATE juce::juce_audio_utils iphlpapi PUBLIC juce::juce_recommended_config_flags)

add_library(MobileBrowser SHARED Source/MobileBrowser.cpp)
set_target_properties(MobileBrowser PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/BrowserRuntime" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/BrowserRuntime")
target_include_directories(MobileBrowser PRIVATE "${CEF_ROOT}")
target_compile_options(MobileBrowser PRIVATE /utf-8)
target_compile_definitions(MobileBrowser PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN CEF_API_VERSION=15000)
target_link_libraries(MobileBrowser PRIVATE "${CEF_ROOT}/Release/libcef.lib" user32 shell32 advapi32)
target_link_libraries(MobileBrowser PRIVATE delayimp)
target_link_options(MobileBrowser PRIVATE /DELAYLOAD:libcef.dll)
add_custom_command(TARGET MobileBrowser POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Release" "$<TARGET_FILE_DIR:MobileBrowser>"
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CEF_ROOT}/Resources" "$<TARGET_FILE_DIR:MobileBrowser>"
 COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/Release/bootstrap.exe" "$<TARGET_FILE_DIR:MobileBrowser>/MobileBrowser.exe")
add_custom_command(TARGET MobileBrowser POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Source/start.html" "$<TARGET_FILE_DIR:MobileBrowser>/start.html")
add_custom_command(TARGET MobileBrowser POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Source/mobile-input.js" "$<TARGET_FILE_DIR:MobileBrowser>/mobile-input.js")
add_custom_command(TARGET MobileBrowser POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Source/passwords.js" "$<TARGET_FILE_DIR:MobileBrowser>/passwords.js")

option(WEBCAPTURE_BUILD_TESTS "Build instrument tests" ON)
if(WEBCAPTURE_BUILD_TESTS)
 juce_add_console_app(MobileTest PRODUCT_NAME "MobileTest")
 target_sources(MobileTest PRIVATE Tests/Smoke.cpp)
 target_compile_definitions(MobileTest PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(MobileTest PRIVATE juce::juce_audio_utils ole32 mmdevapi iphlpapi)
 target_compile_options(MobileTest PRIVATE /utf-8)
endif()
